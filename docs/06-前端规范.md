```tsx
// frontend/src/App.tsx
import { HashRouter, Routes, Route } from 'react-router-dom'
import { WelcomePage, EditorPage, SettingsPage } from './routes'

export default function App() {
    return (
        <HashRouter>
            <Routes>
                <Route path="/" element={<WelcomePage />} />
                <Route path="/editor" element={<EditorPage />} />
                <Route path="/settings" element={<SettingsPage />} />
            </Routes>
        </HashRouter>
    )
}
```

> **注**：使用 HashRouter 是因为 Wails 嵌入式 WebView 对 HTML5 History API 支持不一致。

## 6.4 状态管理（Zustand）

### 6.4.1 Store 设计

```typescript
// frontend/src/stores/resumeStore.ts
import { create } from 'zustand'
import type { Resume } from '../types/resume'

interface ResumeState {
    // 数据
    resume: Resume | null
    isDirty: boolean              // 是否有未保存的修改
    filePath: string | null       // 当前文件路径（null = 未保存过）

    // 预览
    previewHtml: string
    isPreviewLoading: boolean

    // 操作
    setResume: (resume: Resume) => void
    updateField: (path: string, value: unknown) => void
    markDirty: () => void
    markSaved: (filePath: string) => void
    setPreviewHtml: (html: string) => void
}

export const useResumeStore = create<ResumeState>((set, get) => ({
    resume: null,
    isDirty: false,
    filePath: null,
    previewHtml: '',
    isPreviewLoading: false,

    setResume: (resume) => set({ resume, isDirty: false }),

    updateField: (path, value) => {
        const resume = get().resume
        if (!resume) return
        // 通过 path (如 "personal.full_name") 深度设置值
        setByPath(resume, path, value)
        set({ resume: { ...resume }, isDirty: true })
    },

    markDirty: () => set({ isDirty: true }),
    markSaved: (filePath) => set({ isDirty: false, filePath }),
    setPreviewHtml: (html) => set({ previewHtml: html }),
}))
```

```typescript
// frontend/src/stores/templateStore.ts
import { create } from 'zustand'
import type { TemplateMeta } from '../types/template'

interface TemplateState {
    templates: TemplateMeta[]
    activeTemplateId: string | null

    setTemplates: (templates: TemplateMeta[]) => void
    setActiveTemplate: (id: string) => void
    addTemplate: (template: TemplateMeta) => void
    removeTemplate: (id: string) => void
}

export const useTemplateStore = create<TemplateState>((set) => ({
    templates: [],
    activeTemplateId: null,

    setTemplates: (templates) => set({ templates }),
    setActiveTemplate: (id) => set({ activeTemplateId: id }),
    addTemplate: (template) =>
        set((state) => ({ templates: [...state.templates, template] })),
    removeTemplate: (id) =>
        set((state) => ({
            templates: state.templates.filter((t) => t.id !== id),
        })),
}))
```

## 6.5 表单设计

### 6.5.1 表单组件准则

每个数据模块对应一个 Section 组件，使用 React Hook Form + Zod 进行表单管理和校验。

```tsx
// frontend/src/components/editor/PersonalSection.tsx
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'

const personalSchema = z.object({
    full_name: z.string().min(1, '姓名为必填项').max(50),
    email: z.string().email('邮箱格式不正确').optional().or(z.literal('')),
    phone: z.string().max(30).optional().or(z.literal('')),
    location: z.string().max(100).optional().or(z.literal('')),
    website: z.string().url('网址格式不正确').optional().or(z.literal('')),
    job_title: z.string().max(100).optional().or(z.literal('')),
    years_of_exp: z.number().min(0).max(50).optional(),
})

type PersonalFormData = z.infer<typeof personalSchema>

export function PersonalSection() {
    const resume = useResumeStore((s) => s.resume)
    const updateField = useResumeStore((s) => s.updateField)

    const form = useForm<PersonalFormData>({
        resolver: zodResolver(personalSchema),
        defaultValues: resume?.personal ?? {},
    })

    // debounce 300ms 后触发预览刷新
    const debouncedUpdate = useMemo(
        () => debounce((path: string, value: unknown) => {
            updateField(path, value)
        }, 300),
        [updateField]
    )

    return (
        <Form {...form}>
            <form className="space-y-4">
                <FormField
                    control={form.control}
                    name="full_name"
                    render={({ field }) => (
                        <FormItem>
                            <FormLabel>姓名 *</FormLabel>
                            <FormControl>
                                <Input {...field} onChange={(e) => {
                                    field.onChange(e)
                                    debouncedUpdate('personal.full_name', e.target.value)
                                }} />
                            </FormControl>
                            <FormMessage />
                        </FormItem>
                    )}
                />
                {/* ... 更多字段 */}
            </form>
        </Form>
    )
}
```

## 6.6 预览面板

### 6.6.1 预览刷新机制

```
用户编辑字段
    │
    ▼
debounce 300ms (减少渲染频率)
    │
    ▼
调用 Go RenderTemplate(resume, templateId)
    │
    ▼
返回 HTML 字符串
    │
    ▼
更新 Zustand previewHtml
    │
    ▼
PreviewFrame 使用 iframe srcdoc 渲染
```

```tsx
// frontend/src/components/preview/PreviewPanel.tsx
import { useResumeStore } from '../../stores/resumeStore'

export function PreviewPanel() {
    const previewHtml = useResumeStore((s) => s.previewHtml)
    const zoom = useEditorStore((s) => s.zoom)
    const iframeRef = useRef<HTMLIFrameElement>(null)

    // 当 previewHtml 变化时，更新 iframe 内容
    useEffect(() => {
        if (iframeRef.current && previewHtml) {
            const doc = iframeRef.current.contentDocument
            if (doc) {
                doc.open()
                doc.write(previewHtml)
                doc.close()
            }
        }
    }, [previewHtml])

    return (
        <div className="preview-panel" style={{ transform: `scale(${zoom})`, transformOrigin: 'top center' }}>
            <iframe
                ref={iframeRef}
                className="preview-frame w-full bg-white"
                title="简历预览"
                sandbox="allow-same-origin"
            />
            <PageIndicator />
        </div>
    )
}
```

### 6.6.2 预览区域计算

- A4 比例：210:297 ≈ 1:1.414
- 默认显示宽度：约 600px（可随面板拖动调整）
- iframe 内部以 A4 尺寸渲染，外层通过 CSS `transform: scale()` 适配面板宽度

## 6.7 自动保存

```typescript
// frontend/src/hooks/useAutoSave.ts
import { useEffect, useRef } from 'react'
import { useResumeStore } from '../stores/resumeStore'
import { FileService } from '../bindings/resume-craft'

export function useAutoSave(intervalMs: number = 30000) {
    const isDirty = useResumeStore((s) => s.isDirty)
    const resume = useResumeStore((s) => s.resume)
    const lastSavedRef = useRef<number>(Date.now())

    useEffect(() => {
        const timer = setInterval(async () => {
            if (isDirty && resume) {
                try {
                    await FileService.SaveProject('') // 空路径 = 保存到自动保存文件
                    lastSavedRef.current = Date.now()
                } catch (err) {
                    console.error('自动保存失败:', err)
                }
            }
        }, intervalMs)

        return () => clearInterval(timer)
    }, [isDirty, resume, intervalMs])
}
```

## 6.9 Wails v3 运行时封装

Wails v3 前端运行时与 v2 不同：
- 绑定文件路径从 `wailsjs/go/...` 变为 `frontend/bindings/...`
- 事件监听使用 `app.Event.On()` 而非 `EventsOn()`
- Go 方法的 error 返回现在 reject 为 Error 对象而非字符串

```typescript
// frontend/src/lib/wails-runtime.ts

// 从生成绑定导入
import { ResumeService, TemplateService, ExportService, FileService } from '../bindings/resume-craft'
import { app } from '../bindings/resume-craft'

// 统一封装 Go 服务方法调用，添加错误处理和 loading 状态
export async function callBackend<T>(
    fn: () => Promise<T>,
    onError?: (err: Error) => void
): Promise<T | null> {
    try {
        return await fn()
    } catch (err) {
        // v3: err 是 Error 对象，err.message 包含错误信息
        console.error('Backend call failed:', err)
        onError?.(err as Error)
        return null
    }
}

// 事件监听封装（v3 Manager API）
export function listenBackend<T>(
    event: string,
    handler: (data: T) => void
): () => void {
    const cancelFn = app.Event.On(event, handler)
    return cancelFn  // 返回取消监听函数
}

// 导出服务实例（供组件直接使用）
export { ResumeService, TemplateService, ExportService, FileService, app }
```
